\section{引言：为什么需要锁相放大器？}

\begin{frame}{微弱信号检测的挑战}
    \begin{columns}
        \column{0.55\textwidth}
        \begin{block}{背景}
            在物理测量中，信号往往淹没在噪声中。
            \begin{itemize}
                \item \textbf{噪声来源}：热噪声、散粒噪声、$1/f$ 噪声、市电干扰($50\text{ Hz}$)等 。
            \end{itemize}
        \end{block}
        \begin{alertblock}{锁相放大器 (LIA) }
            \begin{itemize}
                \item 利用\textbf{相干检测} (Coherent Detection) 技术 。
                \item 极窄的等效噪声带宽（可达 $0.0025\text{Hz}$）。
            \end{itemize}
        \end{alertblock}
        \column{0.35\textwidth}

\end{frame}

\section{实验原理}

\begin{frame}{相敏检波 (PSD) 原理推导}
    \begin{columns}
        \column{0.45\textwidth}
        \textbf{输入信号}：
        \begin{equation}
            u_i = U_i \sin(\omega_i t + \varphi)
        \end{equation}
        \textbf{参考信号}（方波展开）：
        \begin{equation}
            u_r = \frac{4}{\pi}\sum_{n=0}^{\infty}\frac{1}{2n+1}\sin[(2n+1)\omega_{r}t]
        \end{equation}
        \textbf{PSD输出} ($u_{oPSD} \propto u_i \times u_r$)：
        \begin{align}
            &
            u_{oPSD}
            \notag \\
            \propto
            &
            \sum_{n=0}^{\infty}\frac{1}{2n+1}
            U_i \bigl\{
                \cos \bigl\{
                    \bigl[ (2n+1)\omega_{r} - \omega_i \bigr] t - \varphi
                \bigr\}
            \notag \\
            &
            -
                \cos \bigl\{
                    \bigl[ (2n+1)\omega_{r} + \omega_i \bigr] t + \varphi
                \bigr\}
            \bigr\}
        \end{align}
        $\omega_i = \omega_r$ 时，仅基波项 ($n=0$) 产生直流分量。
        \column{0.45\textwidth}
        \begin{block}{直流输出 $u_{dc}$}
            经低通滤波器后，仅保留直流分量输出：
            \begin{equation}
                u_{dc} = \frac{2}{\pi} U_i \cos \varphi
            \end{equation}
            \begin{itemize}
                \item 正比于输入幅值 $U_i$。
                \item 与相位差 $\varphi$ 有关。
                \item 当 $\varphi = 0$ 时，输出最大。
            \end{itemize}
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}{谐波响应与噪声抑制}
    \begin{columns}[t]
        \column{0.48\textwidth}
        \begin{itemize}
            \item \textbf{奇次谐波响应}：
            由于参考信号是方波，PSD对输入信号的奇次谐波 ($3\omega, 5\omega...$) 也有响应 ：
            \begin{equation}
                u_{dc} = \left(\frac{1}{n}\right) \frac{2}{\pi} U_i \cos\varphi \quad (n=1, 3, 5...)
            \end{equation}
        \end{itemize}
        \column{0.48\textwidth}
        \begin{itemize}
            \item \textbf{噪声抑制}：
            低通滤波器 (LPF) 的时间常数 $\tau = RC$ 决定了等效噪声带宽 $B$ ；$\tau$ 越大，$B$ 越窄，抑制噪声能力越强，但响应速度变慢。
        \end{itemize}
    \end{columns}

\section{实验内容与步骤}

\begin{frame}{1. 参考信号通道特性研究}
    \begin{columns}
        \column{0.9\textwidth}
        \begin{figure}
            \centering
            \includegraphics[width=0.6\textwidth]{Prep/01-reference-signal.png}
        \end{figure}
        % \column{0.6\textwidth}
        \begin{itemize}
            \item \textbf{设置}：先后设置$f=200\text{ Hz}, U_i \approx 70 \text{ mV}$以及$f=1\text{ kHz}, U_i \approx 100 \text{ mV}$。
            \item \textbf{测量}：示波器测量峰峰值 $U_{pp}$ 和周期 $T$，并记录波形示意图；交流电压表测有效值。
            \item \textbf{验证}：正弦波的波峰因子理论值$U_{pp}/U_i \approx 2\sqrt{2} \approx 2.828$。
            \item \textbf{移相观察}：调节移相器 ($0^\circ \sim 270^\circ$)，观察参考信号与输入信号的相位关系。使相位差计显示的位相差分别为 $0^\circ, 90^\circ, 180^\circ, 270^\circ$，画出宽带移相器的输入和同相输出信号的波形。
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{2. 相敏检波器 (PSD) 研究}
    \begin{columns}
        \column{0.4\textwidth}
        \begin{figure}
            \includegraphics[width=0.9\textwidth]{Prep/02-psd-study.png}
            % \tikz\node[draw, minimum width=5cm, minimum height=3cm] {此处插入图片：课上操作指南 图6};
            \caption{PSD研究接线图}
        \end{figure}
        \column{0.5\textwidth}
        \begin{itemize}
            \item \textbf{参数}：交流放大倍数 $K_{\text{AC}}=10$，直流放大倍数 $K_{\text{DC}}=5$，相关器低通滤波时间常数$1\text{ s}$
            \item \textbf{波形观测}：记录 $\varphi=0^\circ, 90^\circ, 180^\circ, 270^\circ$ 时的PSD输出波形。
            \item \textbf{相位特性}：测量相关器输出直流电压大小 $U_{dc}$ 随 $\varphi$ 的变化（每 $20^\circ$ 一测）。
            \item \textbf{验证}：绘制 $U_{dc}/U_i$ vs $\varphi$ 曲线，验证 $U_{dc} / U_i  = \frac{2^{3/2}}{\pi}\cos\varphi$ 关系。
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{3. 谐波响应与噪声抑制}
    \begin{columns}
        \column{0.5\textwidth}
        \begin{block}{3. 谐波响应}
            \begin{itemize}
                \item 利用信号源的“分频输出”，使参考频率为信号频率的 $1/n$。
                \item 测量 $n=1, 2, 3, 4, 5...$ 时的直流输出。
                \item \textbf{预期}：偶次谐波无输出，奇次谐波按 $1/n$ 衰减 。
            \end{itemize}
        \end{block}
        \column{0.4\textwidth}
        \begin{figure}
            \centering

            \caption{谐波响应实验接线图}
        \end{figure}
    \end{columns}
\end{frame}

\begin{frame}{4. 噪声抑制 (抗干扰)}
    \begin{columns}
        \column{0.5\textwidth}
        \begin{block}{4. 噪声抑制 (抗干扰)}
            \begin{itemize}
                \item 输入：$200\text{ Hz}$ 正弦信号 + 外部低频干扰 ($U_n=85\text{ mV}$)。
                \item 改变干扰频率 $f_{noise} = n f_i$，观察 $U_{dc}$ 的波动范围 $\Delta U_{dc}$。
                \item 验证 LIA 对非同步信号的强大抑制能力。
